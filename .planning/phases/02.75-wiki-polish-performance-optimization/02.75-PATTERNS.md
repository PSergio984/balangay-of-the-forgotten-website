## 3. Dynamic Metadata Generation (`app/(site)/wiki/[category]/[slug]/page.tsx`)

**Existing Analog:**
No dynamic metadata export exists on this page currently.

**Target Pattern:**
```tsx
import type { Metadata } from 'next'

export async function generateMetadata({
  params,
}: {
  params: Promise<{ category: string; slug: string }>
}): Promise<Metadata> {
  const { category, slug } = await params
  const payload = await getPayloadInstance()
  const result = await payload.find({
    collection: category as any,
    where: { slug: { equals: slug } },
    limit: 1,
  })
  const doc = result.docs[0] as any
  if (!doc) return {}
  return {
    title: `${doc.name} - Balangay of the Forgotten Wiki`,
    description: `Read about the history, lore, and traits of ${doc.name}.`,
    openGraph: {
      title: doc.name,
      description: `Explore the lore of ${doc.name} in the official wiki.`,
      images: doc.image?.url ? [{ url: doc.image.url }] : [],
    },
  }
}
```
